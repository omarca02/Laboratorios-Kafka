python3 - <<'PY'
import json
import boto3
import pyarrow.parquet as pq
from io import BytesIO
from botocore.config import Config

c = json.load(open("conector-s3-sink.json"))["config"]

s3 = boto3.client(
    "s3",
    endpoint_url="http://127.0.0.1:9000",
    aws_access_key_id=c["aws.access.key.id"],
    aws_secret_access_key=c["aws.secret.access.key"],
    region_name="us-east-1",
    config=Config(s3={"addressing_style": "path"})
)

total = 0
archivos = 0
nits = []

for pagina in s3.get_paginator("list_objects_v2").paginate(
    Bucket="dian-bronce",
    Prefix="topics/jdbc.declaraciones/"
):
    for obj in pagina.get("Contents", []):
        if not obj["Key"].endswith(".parquet"):
            continue

        datos = s3.get_object(
            Bucket="dian-bronce",
            Key=obj["Key"]
        )["Body"].read()

        tabla = pq.read_table(BytesIO(datos))
        cantidad = tabla.num_rows
        archivos += 1
        total += cantidad

        if "nit" in tabla.column_names:
            nits.extend(tabla.column("nit").to_pylist())

        print(f"Archivo {archivos}: {cantidad} registros")

print("\nRESUMEN")
print("Archivos Parquet:", archivos)
print("Total de registros almacenados:", total)
print("Columnas:", tabla.column_names if archivos else [])

if nits:
    print("NIT seudonimizado:", all(
        nit == "SEUDONIMIZADO" for nit in nits
    ))

import json, boto3

from io import BytesIO
from botocore.config import Config

    "s3",

    aws_access_key_id=c["aws.access.key.id"],
