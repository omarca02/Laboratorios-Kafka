#!/usr/bin/env python3
"""Productor Avro para el evento dian.declaracion.presentada."""

import argparse
import random
import uuid
from datetime import datetime, timezone
from decimal import Decimal
from pathlib import Path

from confluent_kafka import SerializingProducer
from confluent_kafka.schema_registry import SchemaRegistryClient
from confluent_kafka.schema_registry.avro import AvroSerializer
from confluent_kafka.serialization import StringSerializer


TOPIC = "dian.declaracion.presentada"
BOOTSTRAP_SERVERS = "localhost:9092"
SCHEMA_REGISTRY_URL = "http://localhost:8081"
SCHEMA_PATH = Path(__file__).with_name("declaracion-presentada-v1.avsc")


def crear_evento(invalido: bool = False) -> dict:
    """Construye un evento sintético que cumple el contrato Avro."""
    valor = Decimal(random.randint(100_000_00, 50_000_000_00)) / Decimal("100")
    return {
        "id_evento": str(uuid.uuid4()),
        "nit": str(random.randint(800_000_000, 999_999_999)),
        "formulario": random.choice(["110", "300", "350", "490"]),
        "periodo": "2026-07",
        "fecha_presentacion": int(datetime.now(timezone.utc).timestamp() * 1000),
        "valor_pagado": valor.quantize(Decimal("0.01")),
        "estado": "EN_TRAMITE" if invalido else random.choice(
            ["PRESENTADA", "ANULADA", "CORREGIDA"]
        ),
    }


def main() -> None:
    parser = argparse.ArgumentParser(description="Produce eventos tributarios en Avro")
    parser.add_argument("--n", type=int, default=50, help="Cantidad de eventos")
    parser.add_argument(
        "--invalido",
        action="store_true",
        help="Intenta producir EN_TRAMITE para comprobar la validación del enum",
    )
    args = parser.parse_args()

    schema_str = SCHEMA_PATH.read_text(encoding="utf-8")
    registry = SchemaRegistryClient({"url": SCHEMA_REGISTRY_URL})
    avro_serializer = AvroSerializer(
        registry,
        schema_str,
        conf={"auto.register.schemas": False},
    )

    producer = SerializingProducer(
        {
            "bootstrap.servers": BOOTSTRAP_SERVERS,
            "key.serializer": StringSerializer("utf_8"),
            "value.serializer": avro_serializer,
            "acks": "all",
            "enable.idempotence": True,
        }
    )

    enviados = 0
    for _ in range(args.n):
        evento = crear_evento(args.invalido)
        try:
            producer.produce(
                topic=TOPIC,
                key=evento["nit"],
                value=evento,
            )
            enviados += 1
            producer.poll(0)
        except Exception as exc:
            print(f"[RECHAZADO POR EL ESQUEMA] {type(exc).__name__}: {exc}")
            break

    producer.flush()
    print(f"enviados: {enviados}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Consumidor Avro para el evento dian.declaracion.presentada."""

import argparse

from confluent_kafka import DeserializingConsumer, KafkaError
from confluent_kafka.schema_registry import SchemaRegistryClient
from confluent_kafka.schema_registry.avro import AvroDeserializer
from confluent_kafka.serialization import StringDeserializer


TOPIC = "dian.declaracion.presentada"
BOOTSTRAP_SERVERS = "localhost:9092"
SCHEMA_REGISTRY_URL = "http://localhost:8081"


def main() -> None:
    parser = argparse.ArgumentParser(description="Consume eventos tributarios en Avro")
    parser.add_argument("--group", required=True, help="Grupo consumidor")
    args = parser.parse_args()

    registry = SchemaRegistryClient({"url": SCHEMA_REGISTRY_URL})
    avro_deserializer = AvroDeserializer(registry)

    consumer = DeserializingConsumer(
        {
            "bootstrap.servers": BOOTSTRAP_SERVERS,
            "group.id": args.group,
            "auto.offset.reset": "earliest",
            "enable.auto.commit": False,
            "key.deserializer": StringDeserializer("utf_8"),
            "value.deserializer": avro_deserializer,
        }
    )

    total = 0
    consumer.subscribe([TOPIC])
    try:
        while True:
            mensaje = consumer.poll(5.0)
            if mensaje is None:
                print(f"sin más mensajes ({total} leídos)")
                break
            if mensaje.error():
                if mensaje.error().code() == KafkaError._PARTITION_EOF:
                    continue
                raise RuntimeError(mensaje.error())

            evento = mensaje.value()
            total += 1
            if total <= 5:
                print(
                    f"p={mensaje.partition()} o={mensaje.offset()} "
                    f"nit={evento['nit']} form={evento['formulario']} "
                    f"valor={evento['valor_pagado']:.2f} estado={evento['estado']}"
                )

        if total:
            consumer.commit(asynchronous=False)
    finally:
        consumer.close()


if __name__ == "__main__":
    main()
